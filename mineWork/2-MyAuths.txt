CREATE TABLE users (
    user_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    roll_number TEXT UNIQUE NOT NULL,  -- Login identifier
    user_name TEXT UNIQUE NOT NULL,    -- Display name
    gender TEXT CHECK (gender IN ('M', 'F', 'O', 'N')) NOT NULL,
    branch TEXT NOT NULL,
    password_hash TEXT NOT NULL,
    dob DATE, -- Editable and Null
    dp_url TEXT DEFAULT '/default-avatar.png',
    is_active BOOLEAN DEFAULT TRUE,
    failed_login_attempts INT DEFAULT 0,
    account_locked_until TIMESTAMP,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    last_seen_at TIMESTAMP DEFAULT NOW(),
    deleted_at TIMESTAMP,
);


CREATE TABLE otp_verifications (
    otp_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
    otp_hash TEXT NOT NULL,
    purpose TEXT NOT NULL CHECK (purpose IN ('signup', 'password_reset')),
    attempts INT DEFAULT 0,
    max_attempts INT DEFAULT 3,
    expires_at TIMESTAMP NOT NULL,
    created_at TIMESTAMP DEFAULT NOW(),
    used_at TIMESTAMP
);

-- Index for fast cleanup
CREATE INDEX idx_otp_expires ON otp_verifications(expires_at);


CREATE TABLE refresh_tokens (
    token_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
    token_hash TEXT NOT NULL,
    family_id UUID NOT NULL,
    expires_at TIMESTAMP NOT NULL,
    revoked BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT NOW()
);

-- Index for user lookups
CREATE INDEX idx_refresh_user ON refresh_tokens(user_id);


CREATE TABLE auth_audit_logs (
    log_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(user_id),
    roll_number TEXT,
    event_type TEXT NOT NULL CHECK (event_type IN (
        'login_success', 
        'login_failed', 
        'account_locked', 
        'password_reset_request',
        'password_reset_success',
        'profile_updated',
        'dp_changed',
        'account_deleted',
        'account_deactivated',
        'user_blocked',
        'user_unblocked'
    )),
    ip_address INET,
    user_agent TEXT,
    created_at TIMESTAMP DEFAULT NOW()
);

    
OTP as:
Rate Limiting - Prevent brute force
IP Tracking - Detect suspicious patterns
OTP Binding - Bind to IP/device
Attempt Counting - Max 3 attempts per OTP
Short Expiry - 2 minutes (optimized for email)
Fast Invalidation - Delete after use or max attempts
Encrypted Storage - Secrets hashed
Activity Logging - Track for anomalies
Device Fingerprinting - Additional validation layer
Fast Email Delivery - Resend API (~200ms)




============== Signup ================

1. User submits signup form
   → Validate: name, gender, roll_number(7 chars, last 5 numeric), 
               branch, password
   → Check: roll_number uniqueness

2. Pre-create user record
   → Insert user with: email_verified=FALSE, is_active=FALSE
   → Hash password (Argon2id/bcrypt cost=12)

3. Generate OTP
   → Create 6-digit numeric OTP
   → Hash OTP (SHA256)
   → Store in otp_verifications table (purpose='signup')
   → Set expiry: NOW() + 2 minutes

4. Send email
   → Send OTP to provided email
   → Include IP address in email for security awareness

5. User verifies OTP
   → Validate: OTP matches hash, not expired, attempts < 3
   → Update user: email_verified=TRUE, is_active=TRUE
   → Mark OTP as used

6. Auto-login after verification
   → Generate JWT access_token (15 min expiry)
   → Generate refresh_token (7 days expiry)
   → Store hashed refresh_token in database
   → Return tokens to user

7. Redirect to profile completion


================== Login ===================

1. User submits credentials
   → Validate roll_number format (7 chars, last 5 numeric)
   → Check rate limiting: max 5 attempts/15 min per IP

2. Find user
   → Query by roll_number
   → Check: is_active=TRUE, account_locked_until < NOW()

3. Verify credentials
   → Compare password_hash
   → If match:
        - Reset failed_login_attempts to 0
        - Update last_seen_at
        - Log successful login
        - Generate tokens
   → If no match:
        - Increment failed_login_attempts
        - If attempts ≥ 5: set account_locked_until = NOW() + 30 min
        - Log failed attempt
        - Return generic error

4. Generate tokens
   → access_token: 15 minutes, contains user_id, roll_number
   → refresh_token: 7 days, random 128 chars
   → Hash refresh_token (SHA256)
   → Store in refresh_tokens table with family_id

5. Return response
   → HTTP-only cookie for refresh_token (secure, sameSite=strict)
   → JSON response with access_token


================== Reset ===================

1. Request reset
   → User submits roll_number
   → Find user by roll_number (must exist and is_active=TRUE)
   → Generate OTP (purpose='password_reset')
   → Store hashed OTP (expires in 2 min)
   → Send email with OTP

2. Verify OTP
   → Validate OTP (3 attempts max)
   → On success: generate reset_token (JWT, 10 min expiry)
   → Return reset_token

3. Reset password
   → Verify reset_token
   → Validate new password strength
   → Update password_hash
   → Invalidate all user's refresh_tokens (force re-login)
   → Log password_reset_success
